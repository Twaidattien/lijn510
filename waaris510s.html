Realtime GPS-simulatie logica (Gecorrigeerde interpolatie) let currentLeg = 0; let progress = 0; const stepsPerLeg = 150; function updateBusTracking() { let start = routePoints[currentLeg]; let end = routePoints[(currentLeg + 1) % routePoints.length]; // CORRECTIE: Expliciete indexering voor Lat [0] en Lng [1] let lat = start[0] + (end[0] - start[0]) * (progress / stepsPerLeg); let lng = start[1] + (end[1] - start[1]) * (progress / stepsPerLeg); busMarker.setLatLng([lat, lng]); // Toon de aankomende halte direct in de UI let nextHalte = haltes[(currentLeg + 1) % haltes.length]; document.getElementById('next-stop').innerText = `Volgende halte: ${nextHalte.name}`; progress++; if (progress > stepsPerLeg) { progress = 0; currentLeg = (currentLeg + 1) % routePoints.length; } setTimeout(updateBusTracking, 40); } // Start de tracking-lus updateBusTracking();
